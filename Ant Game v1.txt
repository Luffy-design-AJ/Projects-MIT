  // ────────────────────────────────────────────────
  // ANT UPDATE — with saturation penalty
  // ────────────────────────────────────────────────
  function saturationPenalty(v) {
    // Nonlinear: returns a score modifier based on local concentration
    // Optimal around 0.5, degrades at extremes
    const opt = devParams.satCap * 0.45;
    if(v < opt) return 1.0;
    const excess = (v - opt) / (devParams.satCap - opt + 0.001);
    return Math.max(devParams.satPen, 1.0 - excess * (1.0 - devParams.satPen));
  }

  function updateAnt(ant, rival=false) {
    const lvl = levels[state.currentLevel];
    const nest = rival ? lvl.rivalNest : (lvl.dualNest ? lvl.nests[ant.nestIndex%lvl.nests.length] : lvl.nest);
    const homeField = rival ? state.rivalPher   : state.homePher;
    const playerField= rival ? state.playerPher2 : state.playerPher;
    const enemyField = rival ? state.playerPher  : state.playerPher2;

    const candidates = [];
    for(let offset=-2; offset<=2; offset++) {
      const d = (ant.dir + offset + dirs8.length) % dirs8.length;
      const [dx,dy] = dirs8[d];
      const nx=ant.x+dx, ny=ant.y+dy;
      if(blocked(nx,ny)) continue;

      let score = 0.34 - Math.abs(offset)*0.1;

      if(ant.carry) {
        const pf = get(homeField,nx,ny);
        score += pf * devParams.carryBonus * saturationPenalty(pf);
        const before = Math.abs(nest.x-ant.x)+Math.abs(nest.y-ant.y);
        const after  = Math.abs(nest.x-nx)  +Math.abs(nest.y-ny);
        score += (before-after)*0.38;
      } else {
        const pf = get(playerField,nx,ny);
        const hf = get(homeField,nx,ny);
        score += pf * 3.2 * saturationPenalty(pf);
        score += hf * 1.2 * saturationPenalty(hf);
        score -= get(enemyField,nx,ny) * 0.4;
        if(pf < 0.04 && hf < 0.03) score += devParams.explore * 0.7;
      }
      score += rand(-devParams.noise, devParams.noise);
      candidates.push({d, nx, ny, score});
    }

    if(!candidates.length) { ant.dir=(ant.dir+4)%dirs8.length; return; }

    candidates.sort((a,b)=>b.score-a.score);
    let chosen = candidates[0];
    if(candidates[1] && Math.random() < 0.2) chosen = candidates[1];
    if(!ant.carry && Math.random() < devParams.explore) {
      const exp = candidates.filter(c=>Math.abs(wrappedDelta(ant.dir,c.d))<=2);
      chosen = exp[Math.floor(rand(0,exp.length))] || chosen;
    }

    const turn = wrappedDelta(ant.dir, chosen.d);
    if(Math.abs(turn)>1 && Math.random()<0.55) {
      ant.dir = (ant.dir+Math.sign(turn)+dirs8.length)%dirs8.length;
      const [dx,dy] = dirs8[ant.dir];
      if(!blocked(ant.x+dx, ant.y+dy)){ ant.x+=dx; ant.y+=dy; }
    } else {
      ant.dir=chosen.d; ant.x=chosen.nx; ant.y=chosen.ny;
    }

    if(ant.carry) {
      if(!rival) stamp(state.homePher, ant.x, ant.y, 1, 0.075, true);
      else       stamp(state.rivalPher, ant.x, ant.y, 1, 0.075, true);
    }

    // Pick up food
    for(const food of state.foods) {
      if(food.amount>0 && !ant.carry && dist2(ant.x,ant.y,food.x,food.y)<=food.r*food.r) {
        ant.carry=1; food.amount-=1; ant.scout=false;
        ant.dir=(ant.dir+4)%dirs8.length;
        break;
      }
    }

    // Deliver
    if(dist2(ant.x,ant.y,nest.x,nest.y)<=nest.r*nest.r && ant.carry) {
      ant.carry=0;
      ant.dir=(ant.dir+Math.floor(rand(-1,2))+dirs8.length)%dirs8.length;
      if(rival) state.rivalDeliveries++; else state.deliveries++;
      state.throughputHistory.push(state.timer);
    }
  }

  // ────────────────────────────────────────────────

  // ────────────────────────────────────────────────
  function diffuse(arr, decay, windX=0, windY=0) {
    const copy = new Float32Array(arr);
    for(let y=1; y<rows-1; y++) {
      for(let x=1; x<cols-1; x++) {
        const i = idx(x,y);
        if(state.solid[i]){ arr[i]=0; continue; }
        const avg = (copy[i]*4 + copy[idx(x+1,y)] + copy[idx(x-1,y)] + copy[idx(x,y+1)] + copy[idx(x,y-1)]) / 8;
        const sx=clamp(Math.round(x-windX),1,cols-2), sy=clamp(Math.round(y-windY),1,rows-2);
        arr[i] = Math.max(copy[idx(sx,sy)]*0.16 + avg*0.84, copy[i]*0.88) * decay;
      }
    }
  }

            return;
          }
          state.pheromoneStrokes++;
          lastStrokePos = key;
        }
      }
      stamp(state.playerPher, cx, cy, r, devParams.stamp, true);
    }
    else if(t === 'pheromone2') stamp(state.playerPher2, cx, cy, r, devParams.stamp, true);
    else if(t === 'eraser') {
      stamp(state.playerPher, cx, cy, r, -1.4);
      stamp(state.playerPher2, cx, cy, r, -1.4);
      stamp(state.homePher, cx, cy, r, -0.5);
      stamp(state.rivalPher, cx, cy, r, -0.5);
      addRect(cx-r, cy-r, r*2+1, r*2+1, 0);
      state.blocks = state.blocks.filter(b=>!(cx>=b.x&&cx<b.x+b.w&&cy>=b.y&&cy<b.y+b.h));
    }
    else if(t === 'block') {
      const s = Math.max(1,r);
      state.blocks.push({x:cx, y:cy, w:s, h:s, user:true});
      addRect(cx, cy, s, s, 1);
    }
    else if(t === 'sugar') {
      // Limited sugar placement
      if(state.foods.filter(f=>f.user).length < 3) {
        state.foods.push({x:cx, y:cy, r:2, amount:40, user:true});
      } else {
        showSatWarn('⚠ Sugar limit reached (max 3)');
      }
    }
  }

  let satWarnTimer = 0;
  function showSatWarn(msg='⚠ Trail saturation — gradients collapsing') {
    const el = document.getElementById('satWarn');
    el.textContent = msg;
    el.classList.add('show');
    satWarnTimer = 2.5;
  }

  // ────────────────────────────────────────────────
  // UPDATE LOOP
  // ────────────────────────────────────────────────
  function update(dt) {
    if(state.paused) return;
    const lvl = levels[state.currentLevel];
    state.timer += dt;

    if(satWarnTimer > 0) {
      satWarnTimer -= dt;
      if(satWarnTimer <= 0) document.getElementById('satWarn').classList.remove('show');
    }

    if(state.mouse.down) applyTool(state.mouse.x, state.mouse.y, state.mouse.right);

    // Compute wind from dev params + level params
    const lp = lvl.params || {};
    const wVarFactor = devParams.windVar > 0 ? (1 + Math.sin(state.timer*0.45)*devParams.windVar/100) : 1;
    const windX = (lp.windX || devParams.windX*0.1) * wVarFactor;
    const windY = (lp.windY || devParams.windY*0.1) * wVarFactor;

    const pDecay = lp.pDecay || devParams.pDecay;
    const hDecay = lp.hDecay || devParams.hDecay;

    diffuse(state.playerPher,  pDecay, windX, windY);
    diffuse(state.playerPher2, pDecay, -windX*0.15, windY*0.15);
    diffuse(state.homePher,    hDecay, windX*0.1, windY*0.1);
    diffuse(state.rivalPher,   hDecay, -windX*0.08, -windY*0.08);

    if(!state.stepAccumulator) state.stepAccumulator = 0;
    state.stepAccumulator += dt;
    const stepRate = 0.088;
    while(state.stepAccumulator >= stepRate) {
      state.ants.forEach(a => updateAnt(a, false));
      state.rivalAnts.forEach(a => updateAnt(a, true));
      state.stepAccumulator -= stepRate;
    }

    state.foods = state.foods.filter(f => f.amount > 0);

    // Saturation check
    let satSum = 0, satCount = 0;
    for(let i=0; i<size; i+=7) {
      satSum += state.playerPher[i];
      satCount++;
    }
    const avgSat = satSum / Math.max(1, satCount);
    const satPct = Math.round(Math.min(100, avgSat / devParams.satCap * 100));
    document.getElementById('satLevel').textContent = satPct + '%';
    if(satPct > 75) showSatWarn();

    // Dev graph
    const density = avgSat;
    densityHistory.shift(); densityHistory.push(density);

    // Win / Lose
    if(state.deliveries >= lvl.target) {
      if(state.currentLevel+2 > state.unlocked) state.unlocked = Math.min(levels.length, state.currentLevel+2);
      buildLevelUI();
      state.paused = true;
      showModal(
        state.currentLevel === levels.length-1 ? 'Notebook Complete' : 'Field Solved',
        state.currentLevel === levels.length-1
          ? 'The notebook closes on a stable proof. Every path was found, not given.'
          : `"${lvl.insight}" — Next field is open.`,
        state.currentLevel < levels.length-1
      );
    }

    if(state.timer >= lvl.time && state.deliveries < lvl.target) {
      state.paused = true;
      showModal('Time Expired', 'The trail design never became self-sustaining. Rewrite the page.', false);
    }

    if(lvl.rival && state.rivalDeliveries >= lvl.target && state.deliveries < lvl.target) {
      state.paused = true;
      showModal('Outcompeted', 'Their attractor stabilized first. Study how their gradient formed.', false);
    }
  }

  // ────────────────────────────────────────────────
  // RENDER
  // ────────────────────────────────────────────────
  function colorFor(v, r, g, b) {
    if(v<=0.018) return null;
    const a = Math.min(0.92, v * 1.1);
    return `rgba(${r},${g},${b},${a})`;
  }

  function drawGrid() {
    ctx.fillStyle = '#2a281e';
    ctx.fillRect(0, 0, canvas.width, canvas.height);
    // Subtle checkerboard
    for(let y=0; y<rows; y++) for(let x=0; x<cols; x++) {
      ctx.fillStyle = ((x+y)%2===0) ? '#2b291f' : '#2a281e';
      ctx.fillRect(x*cell, y*cell, cell, cell);
    }
    // Pheromone fields
    for(let y=0; y<rows; y++) {
      for(let x=0; x<cols; x++) {
        const i = idx(x,y);
        let c = null;
        const pa = state.playerPher[i], pb = state.playerPher2[i];
        const ha = state.homePher[i],   ra = state.rivalPher[i];
        if(pa > 0.018) c = colorFor(pa, 195, 139, 44);
        if(pb > 0.018) { const cb = colorFor(pb, 111, 98, 216); if(!c || pb>pa) c=cb; }
        if(ha > 0.02)  { const ch = colorFor(ha, 232, 194, 93); if(!c || ha>Math.max(pa,pb)) c=ch; }
        if(ra > 0.02)  { const cr = colorFor(ra, 200, 95, 95); if(!c || ra>Math.max(pa,pb,ha)) c=cr; }
        if(c) { ctx.fillStyle=c; ctx.fillRect(x*cell, y*cell, cell, cell); }
      }
    }
  }

  function drawBlocks() {
    ctx.fillStyle = '#66614f';
    state.blocks.forEach(b => ctx.fillRect(b.x*cell, b.y*cell, b.w*cell, b.h*cell));
  }

  function drawFood() {
    state.foods.forEach(f => {
      const ratio = f.amount / (f.startAmount || f.amount || 1);
      ctx.fillStyle = f.user ? '#6ea94a' : '#c49228';
      const s = (f.r*2+1)*cell;
      ctx.fillRect((f.x-f.r)*cell, (f.y-f.r)*cell, s, s);
      // Brightness indicates remaining
      ctx.fillStyle = `rgba(255,245,200,${0.3 + ratio*0.5})`;
      ctx.fillRect(f.x*cell+cell/4, f.y*cell+cell/4, cell/2, cell/2);
    });
  }

  function drawNest(n, label, bg, fg) {
    ctx.fillStyle = bg;
    ctx.fillRect((n.x-n.r)*cell, (n.y-n.r)*cell, (n.r*2+1)*cell, (n.r*2+1)*cell);
    ctx.fillStyle = fg;
    ctx.beginPath();
    ctx.arc(n.x*cell+cell/2, n.y*cell+cell/2, n.r*cell*0.62, 0, Math.PI*2);
    ctx.fill();
    ctx.fillStyle = '#f0ead6';
    ctx.font = `700 ${Math.max(10,cell*1.6)}px IBM Plex Sans`;
    ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
    ctx.fillText(label, n.x*cell+cell/2, n.y*cell+cell/2+1);
  }

  function drawNests() {
    const lvl = levels[state.currentLevel];
    if(lvl.dualNest) {
      drawNest(lvl.nests[0], 'A', '#4a4838', '#aaa277');
      drawNest(lvl.nests[1], 'B', '#284838', '#52c494');
    } else {
      drawNest(lvl.nest, 'N', '#1e3d2a', '#52c494');
    }
    if(lvl.rival && lvl.rivalNest) drawNest(lvl.rivalNest, 'R', '#3d1e1e', '#e07070');
  }

  function drawAnts(list, rival=false) {
    list.forEach(a => {
      if(a.carry) ctx.fillStyle = '#f6c04a';
      else ctx.fillStyle = rival ? '#efb0b0' : '#f0ead4';
      ctx.fillRect(a.x*cell+cell/4, a.y*cell+cell/4, cell/2, cell/2);
    });
  }

  function drawCursor() {
    const colors = { pheromone:'rgba(195,139,44,.9)', pheromone2:'rgba(111,98,216,.9)', block:'rgba(108,104,80,.9)', sugar:'rgba(110,169,74,.9)', eraser:'rgba(180,80,60,.7)' };
    ctx.strokeStyle = colors[state.tool] || 'rgba(195,139,44,.9)';
    ctx.lineWidth = 1;
    ctx.strokeRect(state.mouse.x*cell, state.mouse.y*cell, cell*Math.max(1,state.brush), cell*Math.max(1,state.brush));
  }

  function drawDevGraph() {
    const gc = devGraphCtx;
    const gw = gc.canvas.width, gh = gc.canvas.height;
    gc.fillStyle = '#2a281e';
    gc.fillRect(0,0,gw,gh);
    const max = Math.max(0.01, ...densityHistory);
    gc.strokeStyle = '#c38b2c';
    gc.lineWidth = 1.5;
    gc.beginPath();
    densityHistory.forEach((v,i) => {
      const px = i/densityHistory.length * gw;
      const py = gh - (v/max) * (gh-4) - 2;
      i===0 ? gc.moveTo(px,py) : gc.lineTo(px,py);
    });
    gc.stroke();
    // Saturation cap line
    const capY = gh - (devParams.satCap/max) * (gh-4) - 2;
    gc.strokeStyle = 'rgba(192,57,43,.6)';
    gc.setLineDash([3,3]);
    gc.beginPath(); gc.moveTo(0,capY); gc.lineTo(gw,capY); gc.stroke();
    gc.setLineDash([]);
  }

  function updateUI() {
    const lvl = levels[state.currentLevel];
    const rs = document.getElementById('rivalStatus');
    if(lvl.rival) {
      rs.style.display = '';
      rs.textContent = `Rival: ${state.rivalDeliveries}`;
    } else {
      rs.style.display = 'none';
    }
    document.getElementById('goalStatus').textContent = `${state.deliveries} / ${lvl.target}`;
    document.getElementById('timerStatus').textContent = `${Math.max(0,Math.ceil(lvl.time-state.timer))}s`;
    document.getElementById('foodCollected').textContent = state.deliveries;

    const recent = state.throughputHistory.filter(t=>state.timer-t<15).length;
    document.getElementById('throughput').textContent = `${Math.round(recent*4)}/m`;

    let avg=0, c=0;
    for(let i=0; i<size; i+=29){ avg+=state.playerPher[i]+state.homePher[i]; c++; }
    document.getElementById('trailStrength').textContent = (avg/Math.max(1,c)).toFixed(2);

    // Budget display for lvl 5
    if(lvl.pheromoneBudget !== undefined) {
      document.getElementById('hintText').textContent = `${lvl.hint} [Strokes: ${state.pheromoneStrokes}/${lvl.pheromoneBudget}]`;
    }
  }

  function render() {
    drawGrid(); drawBlocks(); drawFood(); drawNests();
    drawAnts(state.ants, false); drawAnts(state.rivalAnts, true);
    drawCursor(); updateUI();
    // Dev graph only if tab visible
    if(document.getElementById('tab-dev').classList.contains('visible')) drawDevGraph();
  }

  // ────────────────────────────────────────────────
  // LEVEL LOADING
  // ────────────────────────────────────────────────
  function loadLevel(i) {
    state.currentLevel = i;
    allocArrays();
    state.ants=[]; state.rivalAnts=[]; state.foods=[]; state.blocks=[];
    state.deliveries=0; state.rivalDeliveries=0; state.timer=0;
    state.throughputHistory=[]; state.pheromoneStrokes=0; lastStrokePos=null;
    state.stepAccumulator=0;

    const lvl = levels[i];
    state.brush = lvl.brush||1;
    document.getElementById('brushSlider').value = state.brush;
    document.getElementById('brushValue').textContent = state.brush;

    (lvl.walls||[]).forEach(w=>{ state.blocks.push({...w}); addRect(w.x,w.y,w.w,w.h,1); });
    lvl.foods.forEach(f=>{ state.foods.push({...f, startAmount:f.amount}); });

    const antCount = devParams.antCount || lvl.ants;
    for(let j=0;j<antCount;j++) state.ants.push(spawnAnt(j%(lvl.dualNest?lvl.nests.length:1), false));
    for(let j=0;j<(lvl.rivalAnts||0);j++) state.rivalAnts.push(spawnAnt(0, true));

    // Apply level params to devParams if defined
    if(lvl.params) {
      if(lvl.params.pDecay) devParams.pDecay = lvl.params.pDecay;
      if(lvl.params.hDecay) devParams.hDecay = lvl.params.hDecay;
      if(lvl.params.windX !== undefined) devParams.windX = lvl.params.windX * 10;
      if(lvl.params.windY !== undefined) devParams.windY = lvl.params.windY * 10;
      if(lvl.params.windVar !== undefined) devParams.windVar = lvl.params.windVar;
      syncDevSliders();
    }

    state.tool = availableTools()[0];
    refreshToolButtons();
    buildLevelUI();

    document.getElementById('levelTitle').textContent = `${lvl.id}. ${lvl.name}`;
    document.getElementById('levelGoal').textContent = lvl.goal;
    document.getElementById('hintTitle').textContent = {curiosity:'Curiosity', pattern:'Pattern', mastery:'Mastery'}[lvl.phase]||'Observe';
    document.getElementById('hintText').textContent = lvl.hint;
    document.getElementById('levelFlavor').textContent = lvl.flavor;
    document.getElementById('rivalStatus').style.display = 'none';

    state.paused = false;
    document.getElementById('playPause').textContent = '⏸ Pause';
    hideModal();
  }

  function availableTools() { return levels[state.currentLevel].tools; }

  function buildToolUI() {
    const grid = document.getElementById('toolGrid');
    grid.innerHTML = '';
    Object.entries(toolDefs).forEach(([key, def]) => {
      const b = document.createElement('button');
      b.className = 'btn tool-btn';
      b.innerHTML = `${def.label}<span class="t-icon">${def.sub}</span>`;
      b.dataset.tool = key;
      b.onclick = () => {
        if(availableTools().includes(key)) { state.tool=key; refreshToolButtons(); }
      };
      grid.appendChild(b);
    });
    refreshToolButtons();
  }

  function refreshToolButtons() {
    document.querySelectorAll('.tool-btn').forEach(btn => {
      const ok = availableTools().includes(btn.dataset.tool);
      btn.disabled = !ok;
      btn.style.opacity = ok ? 1 : .3;
      btn.classList.toggle('active', state.tool===btn.dataset.tool);
    });
  }

  const phaseLabels = { curiosity: 'Curiosity', pattern: 'Pattern', mastery: 'Mastery' };
  function buildLevelUI() {
    const list = document.getElementById('levelList');
    list.innerHTML = '';
    levels.forEach((lvl,i) => {
      const locked = i+1 > state.unlocked;
      const b = document.createElement('button');
      b.className = 'level-btn' + (i===state.currentLevel?' active':'');
      b.disabled = locked;
      b.style.opacity = locked ? .35 : 1;
      b.innerHTML = `<strong>${lvl.id}. ${lvl.name}</strong><small>${lvl.goal}</small><span class="phase-badge phase-${lvl.phase}">${phaseLabels[lvl.phase]}</span>`;
      b.onclick = () => { if(!locked) loadLevel(i); };
      list.appendChild(b);
    });
  }

    hideModal();

  };
  document.getElementById('modalRetry').onclick = () => {
    hideModal();

  };

  // ────────────────────────────────────────────────
  // INPUT
  // ────────────────────────────────────────────────
  const rectToCell = e => {
    const r = canvas.getBoundingClientRect();
    return {
      x: clamp(Math.floor((e.clientX-r.left)*(canvas.width/r.width)/cell), 0, cols-1),
      y: clamp(Math.floor((e.clientY-r.top)*(canvas.height/r.height)/cell), 0, rows-1)
    };
  };
  canvas.addEventListener('mousemove', e => Object.assign(state.mouse, rectToCell(e)));
  canvas.addEventListener('mousedown', e => { Object.assign(state.mouse, rectToCell(e)); state.mouse.down=true; state.mouse.right=e.button===2; });
  canvas.addEventListener('mouseup', () => { state.mouse.down=false; lastStrokePos=null; });
  canvas.addEventListener('mouseleave', () => state.mouse.down=false);
  canvas.addEventListener('contextmenu', e => e.preventDefault());
  canvas.addEventListener('touchstart', e => { const t=e.touches[0]; Object.assign(state.mouse,rectToCell(t)); state.mouse.down=true; },{passive:true});
  canvas.addEventListener('touchmove', e => { const t=e.touches[0]; Object.assign(state.mouse,rectToCell(t)); },{passive:true});
  canvas.addEventListener('touchend', () => { state.mouse.down=false; lastStrokePos=null; });

  document.getElementById('playPause').onclick = () => {
    state.paused = !state.paused;
    document.getElementById('playPause').textContent = state.paused ? '▶ Resume' : '⏸ Pause';
  };
  document.getElementById('resetLevel').onclick = () => loadLevel(state.currentLevel);
  document.getElementById('themeToggle').onclick = () => setTheme(true);
  document.getElementById('brushSlider').oninput = e => {
    state.brush = +e.target.value;
    document.getElementById('brushValue').textContent = state.brush;
  };
  window.addEventListener('keydown', e => {
    if(e.key===' '){ e.preventDefault(); document.getElementById('playPause').click(); }
    if(e.key.toLowerCase()==='r') loadLevel(state.currentLevel);
  });

  // ────────────────────────────────────────────────
  // BOOTSTRAP
  // ────────────────────────────────────────────────
  buildToolUI();
  setTheme();
  loadLevel(0);

  let last = performance.now();
  function frame(now) {
    const dt = Math.min(0.05, (now-last)/1000);
    last = now;
    update(dt);
    render();
    requestAnimationFrame(frame);
  }
  requestAnimationFrame(frame);
})();
