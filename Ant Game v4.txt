  function stamp(arr, x, y, r, v, permanent=false) {
    for(let yy=y-r; yy<=y+r; yy++)
      for(let xx=x-r; xx<=x+r; xx++) {
        if(xx<0||xx>=cols||yy<0||yy>=rows) continue;
        const d = Math.abs(xx-x)+Math.abs(yy-y);
        if(d<=r) {
          const i = idx(xx,yy);
          const cap = permanent ? devParams.satCap * 1.5 : devParams.satCap;
          arr[i] = clamp(arr[i] + v*(1-d/(r+1)), 0, cap);
        }
      }
  }

  function diffuseFootTrail() {
    const decay = devParams.ftDecay;
    const diff  = devParams.ftDiffusion;  // 0 = no spread, 1 = full spread
    const ft = state.footTrail;
    const copy = new Float32Array(ft);
    for(let y=1; y<rows-1; y++) {
      for(let x=1; x<cols-1; x++) {
        const i = idx(x,y);
        if(state.solid[i]){ ft[i]=0; continue; }
        // Very low diffusion — keeps trail line-like
        const neighbors = copy[idx(x+1,y)] + copy[idx(x-1,y)] + copy[idx(x,y+1)] + copy[idx(x,y-1)];
        const spread = diff * 0.25 * neighbors;
        ft[i] = (copy[i] * (1 - diff) + spread) * decay;
      }
    }
  }

  function saturationPenalty(v) {
    const opt = devParams.satCap * 0.5;
    if (v < opt) return 1.0;
    const excess = (v - opt) / (devParams.satCap - opt + 1e-6);
    return Math.max(devParams.satPen, 1.0 - excess * (1.0 - devParams.satPen));
  }

  function updateAnt(ant, rival = false) {
    const lvl = levels[state.currentLevel];
    const nest = rival
      ? lvl.rivalNest
      : (lvl.dualNest ? lvl.nests[ant.nestIndex % lvl.nests.length] : lvl.nest);

    const homeField   = rival ? state.rivalPher    : state.homePher;
    const playerField = rival ? state.playerPher2  : state.playerPher;
    const enemyField  = rival ? state.playerPher   : state.playerPher2;

    const noise = devParams.noise;
    const prevX = ant.x, prevY = ant.y;

    ant.straightness = ant.straightness ?? 1.0;
    ant.confidence   = ant.confidence   ?? 0;
    ant.prevPf = ant.prevPf ?? 0;
    ant.prevHf = ant.prevHf ?? 0;
    ant.prevFt = ant.prevFt ?? 0;

    const candidates = [];

    for (let offset = -2; offset <= 2; offset++) {
      const d = (ant.dir + offset + dirs8.length) % dirs8.length;
      const [dx, dy] = dirs8[d];
      const nx = ant.x + dx, ny = ant.y + dy;
      if (blocked(nx, ny)) continue;

      const hf  = get(homeField,   nx, ny);
      const pf  = get(playerField, nx, ny);
      const ef  = get(enemyField,  nx, ny);
      const ft  = get(state.footTrail, nx, ny);
      const pfr = get(state.playerFresh, nx, ny);  // freshness of player pheromone at this cell

      const hfS = saturationPenalty(hf);
      const pfS = saturationPenalty(pf);

      // Base directional bias
      let score = 0.3 - Math.abs(offset) * 0.08;

      const dPf = pf - ant.prevPf;
      const dHf = hf - ant.prevHf;
      const dFt = ft - ant.prevFt;

      // ── PLAYER pheromone freshness multiplier ──────────────────────
      // Fresh player pher (pfr near 1.0) gets a large bonus.
      // Stale player pher (pfr near 0) contributes base value only.
      const freshMult = 1.0 + pfr * devParams.playerFreshBonus;
      const playerScore = pf * pfS * freshMult;

      if (!ant.carry) {
        // ════════════════════════════════════════
        // FORAGING MODE
        // ════════════════════════════════════════

        // Player pheromone — dominant when fresh
        score += playerScore * 3.5;

        // Home pheromone — mild foraging attraction
        score += hf * 0.8 * hfS;

        // Enemy pheromone repulsion
        score -= ef * 0.6;

        // Foot trail as highway for foragers (when no strong fresh player pher nearby)
        // Check if any fresh player pher is in local 3-cell radius
        const localFresh = get(state.playerFresh, ant.x, ant.y);
        if (localFresh < 0.15 && ant.everDelivered) {
          // Use foot trail + home trail as highway (medium-high variability added below)
          score += ft  * devParams.foragHwyW;
          score += hf  * devParams.foragHwyW * 0.8;
        }

        if (pf < 0.03 && hf < 0.03 && ft < 0.03) {
          score += devParams.explore * 0.9;
        }

      } else {
        // ════════════════════════════════════════
        // RETURN MODE — new physics
        // ════════════════════════════════════════

        // Home pheromone — primary navigator (strongest)
        // But IGNORE cells in this ant's own freshHomeSet (anti-loop)
        const cellKey = `${nx},${ny}`;
        const isOwnFreshHome = ant.freshHomeSet.has(cellKey);

        const homeContrib = isOwnFreshHome
          ? hf * hfS * 0.05   // nearly ignore own fresh deposits
          : hf * hfS * devParams.homeW;

        score += homeContrib;

        // Foot trail — secondary guide for returning ants
        score += ft * devParams.ftReturnW;

        // Player pheromone — dominant when fresh (overrides if freshness high)
        score += playerScore * 2.8;

        // Gradient tracking
        score += dFt * 1.5 + dHf * 1.2;
        score += rand(-devParams.returnNoise * 0.7, devParams.returnNoise * 0.7);
      }

      // General noise
      score += rand(-noise, noise);

      candidates.push({ d, nx, ny, score });
    }

    ant.prevPf = get(playerField, ant.x, ant.y);
    ant.prevHf = get(homeField,   ant.x, ant.y);
    ant.prevFt = get(state.footTrail, ant.x, ant.y);

    if (!candidates.length) {
      ant.dir = (ant.dir + 4) % dirs8.length;
      return;
    }

    candidates.sort((a, b) => b.score - a.score);
    let chosen = candidates[0];

    // Foragers not on route get medium-high variability via occasional 2nd-choice
    const variabilityChance = (!ant.carry && !ant.everDelivered) ? 0.15
      : (ant.carry ? 0.10 : 0.20);  // returning: lower; foragers reusing highway: higher
    if (candidates[1] && Math.random() < variabilityChance) {
      chosen = candidates[1];
    }

    const turn = wrappedDelta(ant.dir, chosen.d);
    if (Math.abs(turn) > 1 && Math.random() < 0.5) {
      ant.dir = (ant.dir + Math.sign(turn) + dirs8.length) % dirs8.length;
      const [dx, dy] = dirs8[ant.dir];
      if (!blocked(ant.x + dx, ant.y + dy)) { ant.x += dx; ant.y += dy; }
    } else {
      ant.dir = chosen.d;
      ant.x = chosen.nx;
      ant.y = chosen.ny;
    }

    const moved = (ant.x !== prevX || ant.y !== prevY);
    ant.straightness = 0.8 * ant.straightness + 0.2 * (Math.abs(turn) <= 1 ? 1 : 0.3);
    ant.confidence   = 0.9 * ant.confidence   + 0.1 * Math.min(1, Math.max(0, ant.straightness));

    // ── FOOT TRAIL deposit — always, every step ────────────────────
    if (moved && !rival) {
      const ftI = idx(ant.x, ant.y);
      state.footTrail[ftI] = clamp(state.footTrail[ftI] + devParams.ftDeposit, 0, devParams.satCap * 1.5);
    }

    // ── HOME PHEROMONE deposit — returning ants, on confident steps ─
    if (ant.carry && ant.confidence > 0.20 && moved && !rival) {
      const decay = lvl.homeDecay || devParams.hDecay;
      const strength = 0.12 * (decay > 0.999 ? 1.0 : 0.8);
      stamp(state.homePher, ant.x, ant.y, 1, strength);

      // Track own fresh home deposits for anti-loop
      const cellKey = `${ant.x},${ant.y}`;
      ant.freshHomeSet.add(cellKey);
      ant.freshHomeAge++;
      // Prune old entries so set doesn't grow forever (~last 60 steps)
      if (ant.freshHomeAge % 60 === 0) {
        ant.freshHomeSet.clear();
        // Re-seed with current position so loop prevention stays active
        ant.freshHomeSet.add(cellKey);
      }
    } else if (rival && ant.carry && ant.confidence > 0.20 && moved) {
      stamp(state.rivalPher, ant.x, ant.y, 1, 0.10);
    }

    // ── FOOD PICKUP ────────────────────────────────────────────────
    for (const food of state.foods) {
      if (food.amount > 0 && !ant.carry && dist2(ant.x, ant.y, food.x, food.y) <= food.r * food.r) {
        ant.carry = 1;
        food.amount -= 1;
        ant.dir = (ant.dir + 4) % dirs8.length;
        // Clear fresh home set on pickup — start a new return trip
        ant.freshHomeSet.clear();
        ant.freshHomeAge = 0;
        ant.stepsWithoutInteraction = 0;
        break;
      }
    }

    // ── DELIVERY ───────────────────────────────────────────────────
    if (dist2(ant.x, ant.y, nest.x, nest.y) <= nest.r * nest.r && ant.carry) {
      ant.carry = 0;
      ant.dir = (ant.dir + Math.floor(rand(-1, 2)) + dirs8.length) % dirs8.length;
      ant.freshHomeSet.clear();
      ant.freshHomeAge = 0;
      ant.stepsWithoutInteraction = 0;
      ant.everDelivered = true;  // This ant has completed a delivery — now uses highway
      if (rival) state.rivalDeliveries++;
      else state.deliveries++;
      state.throughputHistory.push(state.timer);
    }

    ant.stepsWithoutInteraction++;
  }

  // ────────────────────────────────────────────────
  // DIFFUSION
  // ────────────────────────────────────────────────
  function diffuse(arr, decay, windX=0, windY=0) {
    const copy = new Float32Array(arr);
    for(let y=1; y<rows-1; y++) {
      for(let x=1; x<cols-1; x++) {
        const i = idx(x,y);
        if(state.solid[i]){ arr[i]=0; continue; }
        const avg = (copy[i]*4 + copy[idx(x+1,y)] + copy[idx(x-1,y)] + copy[idx(x,y+1)] + copy[idx(x,y-1)]) / 8;
        const sx=clamp(Math.round(x-windX),1,cols-2), sy=clamp(Math.round(y-windY),1,rows-2);
        arr[i] = Math.max(copy[idx(sx,sy)]*0.16 + avg*0.84, copy[i]*0.88) * decay;
      }
    }
  }

  // Player freshness field — just multiply by decay each tick, no spatial diffusion
  function decayPlayerFresh() {
    const d = devParams.playerAgeDecay;
    const pf = state.playerFresh;
    for (let i = 0; i < size; i++) {
      if (pf[i] > 0.001) pf[i] *= d;
      else pf[i] = 0;
    }
  }

  let lastStrokePos = null;

  function applyTool(cx, cy, right=false) {
    const t = right ? 'eraser' : state.tool;
    const r = state.brush;
    const lvl = levels[state.currentLevel];

    if(t === 'pheromone') {
      if(lvl.pheromoneBudget !== undefined) {
        const key = `${cx},${cy}`;
        if(key !== lastStrokePos) {
          if(state.pheromoneStrokes >= lvl.pheromoneBudget) {
            showSatWarn(`⚠ Stroke budget exhausted (${lvl.pheromoneBudget} max)`);
            return;
          }
          state.pheromoneStrokes++;
          lastStrokePos = key;
        }
      }
      stamp(state.playerPher, cx, cy, r, devParams.stamp, true);
      stampFresh(cx, cy, r);  // NEW: mark freshness where player paints
    }
    else if(t === 'pheromone2') {
      stamp(state.playerPher2, cx, cy, r, devParams.stamp, true);
      stampFresh(cx, cy, r);  // trail B also gets freshness boost
    }
    else if(t === 'eraser') {
      let hitPermanent = false;
      for(let yy=cy-r; yy<=cy+r; yy++)
        for(let xx=cx-r; xx<=cx+r; xx++)
          if(xx>=0&&xx<cols&&yy>=0&&yy<rows&&state.permanent[idx(xx,yy)]===1)
            hitPermanent=true;
      if(hitPermanent) { showSatWarn('⚠ Permanent wall — cannot be erased'); return; }
      if(lvl.eraserBudget !== undefined) {
        if(state.eraserCount >= lvl.eraserBudget) { showSatWarn(`⚠ Eraser budget exhausted (${lvl.eraserBudget} max)`); return; }
        const key = `e${cx},${cy}`;
        if(key !== lastStrokePos) { state.eraserCount++; lastStrokePos = key; }
      }
      stamp(state.playerPher, cx, cy, r, -1.4);
      stamp(state.playerPher2, cx, cy, r, -1.4);
      stamp(state.homePher, cx, cy, r, -0.6);
      stamp(state.rivalPher, cx, cy, r, -0.6);
      stamp(state.footTrail, cx, cy, r, -0.5);   // eraser also partially removes foot trail
      stamp(state.playerFresh, cx, cy, r, -1.0);
      addRect(cx-r, cy-r, r*2+1, r*2+1, 0);
      state.blocks = state.blocks.filter(b=>!(cx>=b.x&&cx<b.x+b.w&&cy>=b.y&&cy<b.y+b.h));
    }
    else if(t === 'block') {
      if(lvl.blockBudget !== undefined && state.blockCount >= lvl.blockBudget) { showSatWarn(`⚠ Block budget exhausted (${lvl.blockBudget} max)`); return; }
      const s = Math.max(1,r);
      state.blocks.push({x:cx, y:cy, w:s, h:s, user:true});
      addRect(cx, cy, s, s, 1);
      state.blockCount++;
    }
    else if(t === 'sugar') {
      if(state.foods.filter(f=>f.user).length < 3) {
        state.foods.push({x:cx, y:cy, r:2, amount:40, user:true});
      } else { showSatWarn('⚠ Sugar limit reached (max 3)'); }
    }
  }

    el.classList.add('show');

  function detectFailures() {
    const ants = state.ants;
    if(!ants.length) return;
    const looping = ants.filter(a => a.stepsWithoutInteraction > 120).length;
    state.failureFlags.loop = looping > ants.length * 0.25;
    const cellDensity = new Map();
    for(const a of ants) { const k=`${Math.round(a.x)},${Math.round(a.y)}`; cellDensity.set(k,(cellDensity.get(k)||0)+1); }
    const maxDensity = Math.max(...cellDensity.values());
    state.failureFlags.congestion = maxDensity > 4;
    const carrying = ants.filter(a=>a.carry).length;
    let avgHome=0, c=0;
    for(let i=0; i<size; i+=30){ avgHome+=state.homePher[i]; c++; }
    avgHome /= Math.max(1,c);
    state.failureFlags['trail-collapse'] = carrying > 3 && avgHome < 0.01;
    let scatterCount=0;
    for(const a of ants) { if(a.carry && a.stepsWithoutInteraction>40) scatterCount++; }
    state.failureFlags['return-scatter'] = scatterCount > ants.length * 0.15;
  }

  // ────────────────────────────────────────────────
  // TIER CALCULATION
  // ────────────────────────────────────────────────
  function calcTier(time, tiers) {
    if(time <= tiers.S) return 'S';
    if(time <= tiers.A) return 'A';
    if(time <= tiers.B) return 'B';
    if(time <= tiers.C) return 'C';
    if(time <= tiers.D) return 'D';
    return 'F';
  }

  // ────────────────────────────────────────────────
  // UPDATE LOOP
  // ────────────────────────────────────────────────
  function update(dt) {
    if(state.paused) return;
    const lvl = levels[state.currentLevel];
    state.timer += dt;

    if(satWarnTimer > 0) {
      satWarnTimer -= dt;
      if(satWarnTimer <= 0) document.getElementById('satWarn').classList.remove('show');
    }

    if(state.mouse.down) applyTool(state.mouse.x, state.mouse.y, state.mouse.right);

    const lp = lvl.params || {};
    const wVarFactor = devParams.windVar > 0 ? (1 + Math.sin(state.timer*0.45)*devParams.windVar/100) : 1;
    const windX = (lp.windX || devParams.windX*0.1) * wVarFactor;
    const windY = (lp.windY || devParams.windY*0.1) * wVarFactor;

    const pDecay = lp.pDecay || devParams.pDecay;
    const hDecay = (lvl.homeDecay || devParams.hDecay) * (devParams.homeDecayMult || 1.0);

    diffuse(state.playerPher,  pDecay, windX, windY);
    diffuse(state.playerPher2, pDecay, -windX*0.15, windY*0.15);
    diffuse(state.homePher,    hDecay, windX*0.1, windY*0.1);
    diffuse(state.rivalPher,   hDecay, -windX*0.08, -windY*0.08);

    // NEW: foot trail diffusion (minimal spread)
    diffuseFootTrail();

    // NEW: player freshness decay (fast, no spatial diffusion)
    decayPlayerFresh();

    if(!state.stepAccumulator) state.stepAccumulator = 0;
    state.stepAccumulator += dt;
    const stepRate = 0.088;
    while(state.stepAccumulator >= stepRate) {
      state.ants.forEach(a => updateAnt(a, false));
      state.rivalAnts.forEach(a => updateAnt(a, true));
      state.stepAccumulator -= stepRate;
    }

    state.foods = state.foods.filter(f => f.amount > 0);

    if(Math.floor(state.timer * 12) % 12 === 0) detectFailures();

    let satSum=0, satCount=0;
    for(let i=0; i<size; i+=7){ satSum+=state.playerPher[i]; satCount++; }
    const avgSat = satSum / Math.max(1, satCount);
    const satPct = Math.round(Math.min(100, avgSat / devParams.satCap * 100));
    document.getElementById('satLevel').textContent = satPct + '%';
    if(satPct > 75) showSatWarn('⚠ Trail saturation — gradients collapsing');

    const tier = calcTier(state.timer, lvl.tiers);
    const tierEl = document.getElementById('currentTier');
    tierEl.textContent = state.deliveries > 0 ? tier : '—';
    tierEl.className = 'tier-label tier-' + (state.deliveries > 0 ? tier : 'F');
    document.getElementById('tierHint').textContent = state.deliveries > 0 ?
      (lvl.tierExplain[tier] || '') : 'Make a delivery to start scoring.';

    const densityHistory_val = avgSat;
    densityHistory.shift(); densityHistory.push(densityHistory_val);

    // Win
    if(state.deliveries >= lvl.target) {
      if(state.currentLevel+2 > state.unlocked) state.unlocked = Math.min(levels.length, state.currentLevel+2);
      buildLevelUI();
      state.paused = true;
      const finalTier = calcTier(state.timer, lvl.tiers);
      const tColors = {S:'#d4af37',A:'#52c494',B:'#6f62d8',C:'#c38b2c',D:'#c0392b',F:'#888'};
      document.getElementById('modalTier').textContent = finalTier + ' Tier';
      document.getElementById('modalTier').style.color = tColors[finalTier]||'#888';
      document.getElementById('modalTierExplain').textContent = lvl.tierExplain[finalTier]||'';
      showModal(
        state.currentLevel === levels.length-1 ? 'Notebook Complete' : 'Field Solved',
        `"${lvl.insight}" — Completed in ${Math.floor(state.timer)}s.`,
        state.currentLevel < levels.length-1
      );
    }

    // Lose — time
    if(state.timer >= lvl.time && state.deliveries < lvl.target) {
      state.paused = true;
      document.getElementById('modalTier').textContent = 'Fail';
      document.getElementById('modalTier').style.color = '#888';
      document.getElementById('modalTierExplain').textContent = `Needed ${lvl.target - state.deliveries} more deliveries.`;
      showModal('Time Expired', 'The trail design never became self-sustaining. Reconsider the system.', false);
    }

    // Lose — rival
    if(lvl.rival && state.rivalDeliveries >= lvl.target && state.deliveries < lvl.target) {
      state.paused = true;
      document.getElementById('modalTier').textContent = 'Fail';
      document.getElementById('modalTier').style.color = '#888';
      document.getElementById('modalTierExplain').textContent = 'Rival colony stabilized first.';
      showModal('Outcompeted', 'Their gradient stabilized before yours could disrupt it.', false);
    }
  }

  function colorFor(v, r, g, b) {
    if(v<=0.018) return null;
    const a = Math.min(0.92, v * 1.1);
    return `rgba(${r},${g},${b},${a})`;
  }

  function drawGrid() {
    ctx.fillStyle = '#2a281e';
    ctx.fillRect(0, 0, canvas.width, canvas.height);
    for(let y=0; y<rows; y++) for(let x=0; x<cols; x++) {
      ctx.fillStyle = ((x+y)%2===0) ? '#2b291f' : '#2a281e';
      ctx.fillRect(x*cell, y*cell, cell, cell);
    }
    for(let y=0; y<rows; y++) {
      for(let x=0; x<cols; x++) {
        const i = idx(x,y);
        let c = null;
        const pa = state.playerPher[i], pb = state.playerPher2[i];
        const ha = state.homePher[i],   ra = state.rivalPher[i];
        const ft = state.footTrail[i];

        // Foot trail — subtle teal, shown only when toggle is on OR when value is high
        if(state.showTrail && ft > 0.015) {
          ctx.fillStyle = `rgba(74,200,212,${Math.min(0.7, ft * 3.5)})`;
          ctx.fillRect(x*cell, y*cell, cell, cell);
        }

        if(pa > 0.018) c = colorFor(pa, 195, 139, 44);
        if(pb > 0.018) { const cb = colorFor(pb, 111, 98, 216); if(!c || pb>pa) c=cb; }
        if(ha > 0.02)  { const ch = colorFor(ha, 143, 191, 74); if(!c || ha>Math.max(pa,pb)) c=ch; }
        if(ra > 0.02)  { const cr = colorFor(ra, 200, 95, 95); if(!c || ra>Math.max(pa,pb,ha)) c=cr; }
        if(c) { ctx.fillStyle=c; ctx.fillRect(x*cell, y*cell, cell, cell); }

        // Player freshness — very subtle white shimmer where player just painted
        const pfr = state.playerFresh[i];
        if(pfr > 0.3) {
          ctx.fillStyle = `rgba(255,255,240,${pfr * 0.18})`;
          ctx.fillRect(x*cell, y*cell, cell, cell);
        }
      }
    }
  }

  function drawBlocks() {
    for(let y=0; y<rows; y++) {
      for(let x=0; x<cols; x++) {
        const i=idx(x,y);
        if(state.solid[i]) {
          ctx.fillStyle = state.permanent[i] ? '#4a4836' : '#8a7c5a';
          ctx.fillRect(x*cell, y*cell, cell, cell);
          if(state.permanent[i]) {
            ctx.fillStyle = 'rgba(0,0,0,0.3)';
            ctx.fillRect(x*cell, y*cell+cell-1, cell, 1);
            ctx.fillRect(x*cell+cell-1, y*cell, 1, cell);
          }
        }
      }
    }
  }

  function drawFood() {
    state.foods.forEach(f => {
      const ratio = f.amount / (f.startAmount || f.amount || 1);
      ctx.fillStyle = f.user ? '#6ea94a' : '#c49228';
      const s = (f.r*2+1)*cell;
      ctx.fillRect((f.x-f.r)*cell, (f.y-f.r)*cell, s, s);
      ctx.fillStyle = `rgba(255,245,200,${0.3 + ratio*0.5})`;
      ctx.fillRect(f.x*cell+cell/4, f.y*cell+cell/4, cell/2, cell/2);
    });
  }

  function drawNest(n, label, bg, fg) {
    ctx.fillStyle = bg;
    ctx.fillRect((n.x-n.r)*cell, (n.y-n.r)*cell, (n.r*2+1)*cell, (n.r*2+1)*cell);
    ctx.fillStyle = fg;
    ctx.beginPath();
    ctx.arc(n.x*cell+cell/2, n.y*cell+cell/2, n.r*cell*0.62, 0, Math.PI*2);
    ctx.fill();
    ctx.fillStyle = '#f0ead6';
    ctx.font = `700 ${Math.max(10,cell*1.6)}px IBM Plex Sans`;
    ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
    ctx.fillText(label, n.x*cell+cell/2, n.y*cell+cell/2+1);
  }

  function drawNests() {
    const lvl = levels[state.currentLevel];
    if(lvl.dualNest) {
      drawNest(lvl.nests[0], 'A', '#4a4838', '#aaa277');
      drawNest(lvl.nests[1], 'B', '#284838', '#52c494');
    } else {
      drawNest(lvl.nest, 'N', '#1e3d2a', '#52c494');
    }
    if(lvl.rival && lvl.rivalNest) drawNest(lvl.rivalNest, 'R', '#3d1e1e', '#e07070');
  }

  function drawAnts(list, rival=false) {
    list.forEach(a => {
      const looping = a.stepsWithoutInteraction > 120;
      if(a.carry) ctx.fillStyle = looping ? '#ff8800' : '#f6c04a';
      else ctx.fillStyle = rival ? '#efb0b0' : (looping ? '#ff6666' : '#f0ead4');
      ctx.fillRect(a.x*cell+cell/4, a.y*cell+cell/4, cell/2, cell/2);
    });
  }

  function drawFailureOverlay() {
    const flags = state.failureFlags;
    const active = [];
    if(flags.loop) active.push('⟳ Ant loops detected');
    if(flags.congestion) active.push('⊕ Congestion point');
    if(flags['trail-collapse']) active.push('↯ Home trail collapsed');
    if(flags['return-scatter']) active.push('✦ Return path scatter');
    if(flags.interference) active.push('≈ Signal interference');
    const el = document.getElementById('failureIndicators');
    if(active.length) {
      el.innerHTML = active.map(s=>`<span style="color:var(--alarm)">${s}</span>`).join(' · ');
      document.getElementById('failureBadge').textContent = active[0];
      document.getElementById('failureBadge').classList.add('show');
    } else {
      el.textContent = '';
      document.getElementById('failureBadge').classList.remove('show');
    }
  }

  function drawCursor() {
    const colors = { pheromone:'rgba(195,139,44,.9)', pheromone2:'rgba(111,98,216,.9)', block:'rgba(108,104,80,.9)', sugar:'rgba(110,169,74,.9)', eraser:'rgba(180,80,60,.7)' };
    ctx.strokeStyle = colors[state.tool] || 'rgba(195,139,44,.9)';
    ctx.lineWidth = 1;
    ctx.strokeRect(state.mouse.x*cell, state.mouse.y*cell, cell*Math.max(1,state.brush), cell*Math.max(1,state.brush));
  }

  function drawDevGraph() {
    const gc = devGraphCtx;
    const gw = gc.canvas.width, gh = gc.canvas.height;
    gc.fillStyle = '#2a281e';
    gc.fillRect(0,0,gw,gh);
    const max = Math.max(0.01, ...densityHistory);
    gc.strokeStyle = '#c38b2c';
    gc.lineWidth = 1.5;
    gc.beginPath();
    densityHistory.forEach((v,i) => {
      const px = i/densityHistory.length * gw;
      const py = gh - (v/max) * (gh-4) - 2;
      i===0 ? gc.moveTo(px,py) : gc.lineTo(px,py);
    });
    gc.stroke();
    const capY = gh - (devParams.satCap/max) * (gh-4) - 2;
    gc.strokeStyle = 'rgba(192,57,43,.6)';
    gc.setLineDash([3,3]);
    gc.beginPath(); gc.moveTo(0,capY); gc.lineTo(gw,capY); gc.stroke();
    gc.setLineDash([]);
  }

  function loadLevel(i) {
    state.currentLevel = i;
    allocArrays();
    state.ants=[]; state.rivalAnts=[]; state.foods=[]; state.blocks=[];
    state.deliveries=0; state.rivalDeliveries=0; state.timer=0;
    state.throughputHistory=[]; state.pheromoneStrokes=0;
    state.blockCount=0; state.eraserCount=0;
    lastStrokePos=null;
    state.stepAccumulator=0;
    state.failureFlags = { loop:false, congestion:false, 'trail-collapse':false, interference:false, 'ghost-attractor':false, 'wind-scatter':false, 'wrong-nest':false, 'corner-bounce':false, 'return-scatter':false };

    const lvl = levels[i];
    state.brush = lvl.brush||1;
    document.getElementById('brushSlider').value = state.brush;
    document.getElementById('brushValue').textContent = state.brush;

    (lvl.permanentWalls||[]).forEach(w => { state.blocks.push({...w, permanent:true}); addRect(w.x,w.y,w.w,w.h,1,true); });
    (lvl.softWalls||[]).forEach(w => { state.blocks.push({...w, permanent:false}); addRect(w.x,w.y,w.w,w.h,1,false); });
    (lvl.walls||[]).forEach(w => { state.blocks.push({...w}); addRect(w.x,w.y,w.w,w.h,1); });

    lvl.foods.forEach(f => { state.foods.push({...f, startAmount:f.amount}); });

    const antCount = devParams.antCount || lvl.ants;
    for(let j=0;j<antCount;j++) state.ants.push(spawnAnt(j%(lvl.dualNest?lvl.nests.length:1), false));
    for(let j=0;j<(lvl.rivalAnts||0);j++) state.rivalAnts.push(spawnAnt(0, true));

    if(lvl.params) {
      if(lvl.params.pDecay) devParams.pDecay = lvl.params.pDecay;
      if(lvl.params.hDecay) devParams.hDecay = lvl.params.hDecay;
      if(lvl.params.windX !== undefined) devParams.windX = lvl.params.windX * 10;
      if(lvl.params.windY !== undefined) devParams.windY = lvl.params.windY * 10;
      if(lvl.params.windVar !== undefined) devParams.windVar = lvl.params.windVar;
      syncDevSliders();
    }

    state.tool = availableTools()[0];
    refreshToolButtons();
    buildLevelUI();

    document.getElementById('levelTitle').textContent = `${lvl.id}. ${lvl.name}`;
    document.getElementById('levelGoal').textContent = lvl.goal;
    document.getElementById('hintTitle').textContent = {curiosity:'Curiosity', pattern:'Pattern', mastery:'Mastery'}[lvl.phase]||'Observe';
    document.getElementById('hintText').textContent = lvl.hint;
    document.getElementById('levelFlavor').textContent = lvl.flavor;
    document.getElementById('rivalStatus').style.display = 'none';

    document.getElementById('currentTier').textContent = '—';
    document.getElementById('currentTier').className = 'tier-label tier-F';
    document.getElementById('tierHint').textContent = 'Make a delivery to start scoring.';

    state.paused = false;
    document.getElementById('playPause').textContent = '⏸ Pause';
    hideModal();
  }
